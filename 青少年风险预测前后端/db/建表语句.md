## 1. t_ai_script — AI 话术库

- **用途**：面向 6-12/13-15/16-18 三个年龄段的引导话术，按风险域/类型/等级/平台检索调用。
- **关系**：related_case_id→t_case；related_tag_id→t_risk_label
- **现有数据**：201 行

```sql
DROP TABLE IF EXISTS `t_ai_script`;
CREATE TABLE `t_ai_script` (
  `script_id` varchar(20) NOT NULL COMMENT '话术编号（主键）',
  `related_case_id` varchar(20) DEFAULT NULL COMMENT '关联案例ID → t_case',
  `related_tag_id` varchar(20) DEFAULT NULL COMMENT '关联标签ID → t_risk_label',
  `risk_domain_l1` varchar(50) DEFAULT NULL COMMENT '一级风险域',
  `risk_type_l2` varchar(100) DEFAULT NULL COMMENT '二级风险类型',
  `core_keywords` varchar(500) DEFAULT NULL COMMENT '核心关键词',
  `applicable_platform` varchar(100) DEFAULT NULL COMMENT '适用平台',
  `age_group` varchar(20) DEFAULT NULL COMMENT '适用年龄段（6-12/13-15/16-18）',
  `applicable_risk_level` varchar(20) DEFAULT NULL COMMENT '适用风险等级',
  `script_type` varchar(50) DEFAULT NULL COMMENT '话术类型',
  `script_phase` varchar(50) DEFAULT NULL COMMENT '话术阶段',
  `trigger_condition` text COMMENT '触发条件',
  `script_content` text NOT NULL COMMENT '话术正文',
  `follow_up_action` varchar(500) DEFAULT NULL COMMENT '后续动作',
  `review_status` tinyint(1) DEFAULT '0' COMMENT '审核状态',
  `use_count` int DEFAULT '0' COMMENT '使用次数',
  `effective_rate` decimal(5,2) DEFAULT NULL COMMENT '有效率',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`script_id`),
  KEY `idx_tag_id` (`related_tag_id`),
  KEY `idx_risk_domain` (`risk_domain_l1`),
  KEY `idx_age_group` (`age_group`),
  KEY `idx_risk_level` (`applicable_risk_level`),
  KEY `idx_review_status` (`review_status`),
  KEY `idx_retrieval` (`risk_domain_l1`,`age_group`,`applicable_risk_level`),
  FULLTEXT KEY `ft_script` (`core_keywords`,`trigger_condition`,`script_content`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='AI话术库';
```

## 2. t_case — 案例库

- **用途**：同主题高互动语料聚类而成的典型案例，含传播机制研判与分龄处置策略。
- **关系**：related_tag_id→t_risk_label；representative_corpus_id→t_corpus；related_script_id→t_ai_script
- **现有数据**：60 行

```sql
DROP TABLE IF EXISTS `t_case`;
CREATE TABLE `t_case` (
  `case_id` varchar(20) NOT NULL COMMENT '案例编号（主键）',
  `case_name` varchar(300) NOT NULL COMMENT '案例名称',
  `related_tag_id` varchar(20) DEFAULT NULL COMMENT '关联标签ID → t_risk_label',
  `risk_domain_l1` varchar(50) DEFAULT NULL COMMENT '一级风险域',
  `risk_type_l2` varchar(100) DEFAULT NULL COMMENT '二级风险类型',
  `platform` varchar(50) DEFAULT NULL COMMENT '传播平台',
  `core_keywords` varchar(500) DEFAULT NULL COMMENT '核心关键词',
  `related_corpus_count` int DEFAULT '0' COMMENT '关联语料数量',
  `total_interaction` bigint DEFAULT '0' COMMENT '总互动量',
  `avg_risk_score` decimal(6,2) DEFAULT '0.00' COMMENT '平均风险分',
  `case_priority_score` decimal(6,2) DEFAULT '0.00' COMMENT '案例优先级分',
  `representative_corpus_id` varchar(30) DEFAULT NULL COMMENT '代表语料ID → t_corpus',
  `representative_text` text COMMENT '代表文本',
  `spread_mechanism` text COMMENT '传播机制研判',
  `disposal_strategy_platform` text COMMENT '处置策略（平台侧）',
  `disposal_strategy_6_12` text COMMENT '处置策略（6-12岁）',
  `disposal_strategy_13_15` text COMMENT '处置策略（13-15岁）',
  `disposal_strategy_16_18` text COMMENT '处置策略（16-18岁）',
  `convertible_script_type` varchar(200) DEFAULT NULL COMMENT '可转化话术类型',
  `review_status` tinyint(1) DEFAULT '0' COMMENT '复核状态',
  `related_script_id` varchar(20) DEFAULT NULL COMMENT '关联话术ID → t_ai_script',
  `risk_level` varchar(10) DEFAULT NULL COMMENT '风险等级',
  `incident_time` date DEFAULT NULL COMMENT '发生时间',
  `source_url` varchar(1000) DEFAULT NULL COMMENT '数据来源链接',
  `spread_path` varchar(500) DEFAULT NULL COMMENT '传播路径',
  `interaction_feature` varchar(500) DEFAULT NULL COMMENT '互动特征',
  `review_conclusion` text COMMENT '复盘结论',
  `created_at` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`case_id`),
  KEY `idx_tag_id` (`related_tag_id`),
  KEY `idx_risk_domain` (`risk_domain_l1`),
  KEY `idx_risk_level` (`risk_level`),
  KEY `idx_priority` (`case_priority_score`),
  FULLTEXT KEY `ft_case` (`case_name`,`core_keywords`,`representative_text`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='案例库';
```

```sql
  `publish_time` datetime DEFAULT NULL COMMENT '发布时间',

  `review_status` tinyint(1) DEFAULT '0' COMMENT '复核状态 0/1',

  KEY `idx_platform` (`platform`),
  KEY `idx_risk_level` (`risk_level`),
```


## 13. t_risk_label — 语料标签体系（规则+关键词库 · 四库中枢）

- **用途**：把「关键词→风险域/二级类型/三级标签/风险分/引导方向」的业务规则固化成表，供语料标注、案例归类、话术调用统一引用。
- **关系**：被 t_corpus / t_case / t_ai_script 引用
- **现有数据**：259 行

```sql
DROP TABLE IF EXISTS `t_risk_label`;
CREATE TABLE `t_risk_label` (
  `tag_id` varchar(20) NOT NULL COMMENT '标签编号（主键，如 TAG-00001）',
```
